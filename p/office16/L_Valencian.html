Set it on a lab device (in production use Group Policy or Intune):

$ErrorActionPreference = 'Stop'
$key = 'HKCU:\Software\policies\microsoft\office\16.0\common\languageresources\enabledlanguages'
if (-not (Test-Path $key)) { New-Item -Path $key -Force | Out-Null }
New-ItemProperty -Path $key -Name 'ca-es-valencia' -PropertyType DWord -Value 3 -Force | Out-Null   # Enabled
