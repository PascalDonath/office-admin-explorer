PowerShell

Aktuellen Wert prüfen (nur lesen)

# --- Settings ---
$root = 'HKLM:'   # needs an elevated session

$key = "$root\Software\microsoft\internet explorer\main\featurecontrol\feature_securityband"
foreach ($name in @('groove.exe', 'excel.exe', 'mspub.exe', 'powerpnt.exe', 'pptview.exe', 'visio.exe', 'winproj.exe', 'winword.exe', 'outlook.exe', 'spdesign.exe', 'exprwd.exe', 'msaccess.exe', 'onenote.exe', 'mse7.exe')) {
    try   { '{0} = {1}' -f $name, (Get-ItemPropertyValue -Path $key -Name $name -ErrorAction Stop) }
    catch { '{0}: not configured' -f $name }
}

Adminrechte nötig: HKLM-Werte brauchen eine PowerShell mit erhöhten Rechten.

Aktivieren

# --- Settings ---
$root = 'HKLM:'   # needs an elevated session

try {
    $key = "$root\Software\microsoft\internet explorer\main\featurecontrol\feature_securityband"
    if (-not (Test-Path $key)) { New-Item -Path $key -Force -ErrorAction Stop | Out-Null }
    New-ItemProperty -Path $key -Name 'groove.exe' -PropertyType DWord -Value 1 -Force -ErrorAction Stop | Out-Null   # Enabled
    New-ItemProperty -Path $key -Name 'excel.exe' -PropertyType DWord -Value 1 -Force -ErrorAction Stop | Out-Null   # Enabled
    New-ItemProperty -Path $key -Name 'mspub.exe' -PropertyType DWord -Value 1 -Force -ErrorAction Stop | Out-Null   # Enabled
    New-ItemProperty -Path $key -Name 'powerpnt.exe' -PropertyType DWord -Value 1 -Force -ErrorAction Stop | Out-Null   # Enabled
    New-ItemProperty -Path $key -Name 'pptview.exe' -PropertyType DWord -Value 1 -Force -ErrorAction Stop | Out-Null   # Enabled
    New-ItemProperty -Path $key -Name 'visio.exe' -PropertyType DWord -Value 1 -Force -ErrorAction Stop | Out-Null   # Enabled
    New-ItemProperty -Path $key -Name 'winproj.exe' -PropertyType DWord -Value 1 -Force -ErrorAction Stop | Out-Null   # Enabled
    New-ItemProperty -Path $key -Name 'winword.exe' -PropertyType DWord -Value 1 -Force -ErrorAction Stop | Out-Null   # Enabled
    New-ItemProperty -Path $key -Name 'outlook.exe' -PropertyType DWord -Value 1 -Force -ErrorAction Stop | Out-Null   # Enabled
    New-ItemProperty -Path $key -Name 'spdesign.exe' -PropertyType DWord -Value 1 -Force -ErrorAction Stop | Out-Null   # Enabled
    New-ItemProperty -Path $key -Name 'exprwd.exe' -PropertyType DWord -Value 1 -Force -ErrorAction Stop | Out-Null   # Enabled
    New-ItemProperty -Path $key -Name 'msaccess.exe' -PropertyType DWord -Value 1 -Force -ErrorAction Stop | Out-Null   # Enabled
    New-ItemProperty -Path $key -Name 'onenote.exe' -PropertyType DWord -Value 1 -Force -ErrorAction Stop | Out-Null   # Enabled
    New-ItemProperty -Path $key -Name 'mse7.exe' -PropertyType DWord -Value 1 -Force -ErrorAction Stop | Out-Null   # Enabled
    'Done.'
}
catch { Write-Error "Failed: $($_.Exception.Message)" }
